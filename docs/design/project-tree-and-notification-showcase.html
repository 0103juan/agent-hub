const getFileColor = (name: string) => {
  const ext = name.slice(name.lastIndexOf('.')).toLowerCase();
  switch (ext) {
    case '.ts':
    case '.tsx':
      return 'text-cyan-400';
    case '.js':
    case '.jsx':
      return 'text-amber-400';
    case '.json':
      return 'text-amber-300';
    case '.css':
      return 'text-rose-400';
    default:
      return 'text-slate-400';
  }
};
